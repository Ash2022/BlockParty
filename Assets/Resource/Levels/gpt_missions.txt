i have a unity game - the game is a mix between block tetris and queue mechanics.
you have a grid, sized 3*3 to 9*9 - when game starts you get 3 tetris shapes color coded and 
you place them on the board - completing a row or column removes the row or column from the grid.
in addition there is 1 or more queue of people - they are color coded same like the shapes.
the people at the top of the queue can walk to the tetris board and help the shapes complete 
a row or column, the people walk on their own and only walk if they can help complete a row
or a column, so they dont comlete empty rows or column - only assisting the shapes.

this is the main class of the game

    private void BuildLevel(int levelIndex)
    {
        currLevelData = _modelManager.levelsData[currLevelIndex];

        gameBoard = new int[currLevelData.Rows, currLevelData.Columns];

        for (int i = 0; i < currLevelData.Rows; i++)
        {
            for (int j = 0; j < currLevelData.Columns; j++)
                gameBoard[i, j] = -1;
        }

        //init grid
        _gameGrid.CreateGrid(currLevelData.Rows, currLevelData.Columns);

        List<List<int>> linesModelCopy = new List<List<int>>();

        List<int> numColorsInQueue = new List<int>();

        for (int i = 0; i < currLevelData.PeopleQueues.Count; i++)
        {
            List<int> singleLine = new List<int>();

            for (int j = 0; j < currLevelData.PeopleQueues[i].Count; j++)
            {
                int personColorValue = currLevelData.PeopleQueues[i][j];

                singleLine.Add(personColorValue);

                if (!numColorsInQueue.Contains(personColorValue))
                    numColorsInQueue.Add(personColorValue);
            }

            linesModelCopy.Add(singleLine);
            
        }

        //count how many colors are in the lines 

        currLevelData.NumberOfColors = numColorsInQueue.Count;

        //init lines
        _queueLines.BuildLines(linesModelCopy, _gameGrid, numColorsInQueue.Count);
        //init storage

        shapeCompeltedCounter = 0;

        List<ShapeInfo> shapeInfo = new List<ShapeInfo>();

        shapeInfo = currLevelData.ShapesLines.GetRange(0, 3);//take first 3 shapes

        _shapeStorage.GenerateNewShapes(shapeInfo);
        //start playing

    }

    

    public void RequestNewShapes()
    {
        
        shapeCompeltedCounter++;
        List<ShapeInfo> shapeInfo = new List<ShapeInfo>();

        if(currLevelData.ShapesLines.Count >= (shapeCompeltedCounter+1)*3)
        {
            shapeInfo = currLevelData.ShapesLines.GetRange(shapeCompeltedCounter*3, 3);
            _shapeStorage.GenerateNewShapes(shapeInfo);
        }
        else
        {
            //no more defained shapes - take random ones instead

            bool[] shapesToUse = new bool[4];
            shapesToUse[0] = currLevelData.Shapes1And2;
            shapesToUse[1] = currLevelData.Shapes3;
            shapesToUse[2] = currLevelData.Shapes4;
            shapesToUse[3] = currLevelData.Shapes5;

            _shapeStorage.GenerateRandomShapes(shapesToUse, currLevelData.NumberOfColors, currLevelData.RandomSeed);
        }
    }

    internal int GetNumValidAttemptsToPosition()
    {
        int counter = 0;
        foreach (GridSquareView gridSquareView in _gameGrid.GridSquareViews)
            if (gridSquareView.ValidAttempt)
                counter++;

        return counter;
    }

    public void PositionCurrentShape(ShapeView shape, int colorIndex)
    {
        foreach (GridSquareView gridSquareView in _gameGrid.GridSquareViews)
            if (gridSquareView.ValidAttempt)
            {
                gridSquareView.SetToFull(colorIndex);
                GameBoard[gridSquareView.row, gridSquareView.col] = colorIndex;
            }

        bool shapesListEmpty = _shapeStorage.ShapeWasPositioned(shape);

        PostShapePositionedFlow(shapesListEmpty);

       
    }

    private async void PostShapePositionedFlow(bool shapeListEmpty)
    {
        //check line completions
        RemoveCompletedLines();
        //check line completion with people assist

        bool personAssist = true;

        lockDrag = true;

        while (personAssist)
        {
            personAssist = await CheckPotentialLines();

            _queueLines.OptemizeLine();

        }

        lockDrag = false;

        if (shapeListEmpty)
            RequestNewShapes();

        //check game over complete or fail


        bool allLinesEmpty = _queueLines.AllLinesEmpty();

        if (allLinesEmpty)
            Debug.Log("GameOver - WIN");

        bool noMoreMoves = IsNoMoreMoves();

        if (noMoreMoves)
            Debug.Log("GameOver - NO MORE MOVES");


        if (allLinesEmpty || noMoreMoves)
        {
            //gameOver = true;
            //MenuClicked();

            //_infoText.text = allLinesEmpty ? "LevelComplete" : "LevelFailed";

            GameManager.Instance.GameOver(allLinesEmpty);
        }
    }


    private async Task<bool> CheckPotentialLines()
    {
        Debug.Log("Checking potentialLines");

        int[] currentLineTopByIndex = _queueLines.GetQueueTop();

        //the array contains the amount of each color (based on color index in the array) 

        //for each entry in the array - if the entry is bigger than 0 - see if there is a row that can be completed with this help

        for (int i = 0; i < currentLineTopByIndex.Length; i++)
        {
            if (currentLineTopByIndex[i] > 0)
            {
                List<Vector2Int> resultingIndexes = CheckLineCompletionWithHelp(i, currentLineTopByIndex[i]);

                int totalPeopleToMove = resultingIndexes.Count;

                if (resultingIndexes.Count > 0)
                {
                    for (int j = 0; j < resultingIndexes.Count; j++)
                    {
                        _gameGrid.GridSquareViews[resultingIndexes[j].x, resultingIndexes[j].y].SetToFull(i);
                        gameBoard[resultingIndexes[j].x, resultingIndexes[j].y] = i;

                        //_gridSquareViews[resultingIndexes[j]].ActivateSquere(i);

                        //we have a match - need to pick person of the i color and move to the first index
                        _queueLines.MoveColorPersonToIndex(i, resultingIndexes[j], (person) =>
                        {
                            //move completed
                            totalPeopleToMove--;

                            Destroy(person);                            

                            if (totalPeopleToMove == 0)
                            {
                                Debug.Log("Move Complete");

                                //clear the line on the board
                                RemoveCompletedLines();
                            }
                        });

                    }

                    await Task.Delay(1000);

                    return true;
                }
            }

        }

        return false;
    }

    public void RemoveCompletedLines()
    {
        List<int[]> lines = new List<int[]>();

        List<int> completedRowIndexese = new List<int>();

        //check rows
        for (int row = 0; row < currLevelData.Rows; row++)
        {
            bool rowCompleted = true;
            int rowColorIndex = gameBoard[row, 0];

            if(rowColorIndex !=-1)//empty
            {
                for (int col = 0;col<currLevelData.Columns;col++)
                {
                    if (gameBoard[row,col] != rowColorIndex)
                        rowCompleted = false;
                }
            }
            else
                rowCompleted = false;

            if (rowCompleted)
                completedRowIndexese.Add(row);

        }

        for (int i = 0; i < completedRowIndexese.Count; i++)
        {
            for (int col = 0; col < currLevelData.Columns; col++)
            {
                int rowIndex = completedRowIndexese[i];

                gameBoard[rowIndex, col] = -1;
                _gameGrid.GridSquareViews[rowIndex, col].ClearSquere();
                

            }
        }
        lines = new List<int[]>();

        completedRowIndexese = new List<int>();

        //check cols
        for (int col = 0; col < currLevelData.Columns; col++)
        {
            bool colCompleted = true;
            int colColorIndex = gameBoard[0, col];

            if (colColorIndex != -1)//empty
            {
                for (int row= 0; row < currLevelData.Rows; row++)
                {
                    if (gameBoard[row, col] != colColorIndex)
                        colCompleted = false;
                }
            }
            else
                colCompleted = false;

            if (colCompleted)
                completedRowIndexese.Add(col);

        }

        for (int i = 0; i < completedRowIndexese.Count; i++)
        {
            for (int row = 0; row < currLevelData.Rows; row++)
            {
                int colIndex = completedRowIndexese[i];

                gameBoard[row, colIndex] = -1;
                _gameGrid.GridSquareViews[row, colIndex].ClearSquere();
            }
        }
        return;

    }
    
    private List<Vector2Int> CheckLineCompletionWithHelp(int colorIndex, int _amountOfColor)
    {
        List<Vector2Int> potentialIndexes = new List<Vector2Int>();

        int amountOfColor = Math.Min(currLevelData.Columns - 1, _amountOfColor);

        //check rows
        for (int row = 0; row < currLevelData.Rows; row++)
        {
            potentialIndexes.Clear();
            int rowSqueresMatchingCondition = 0;
            int rowColorIndex = colorIndex;
            bool rowValid = true;

            for (int col = 0; col < currLevelData.Columns; col++)
            {
                int squereColorValue = gameBoard[row,col];

                //the row has at least 1 color that doesnt match
                if (squereColorValue != -1 && squereColorValue != rowColorIndex)
                    rowValid = false;

                if (squereColorValue == rowColorIndex)
                    rowSqueresMatchingCondition++;

                //empty - add to list - maybe we can complete it
                if (squereColorValue == -1)
                    potentialIndexes.Add(new Vector2Int(row,col));


            }

            //row has 1 color or empty spaces
            if (rowValid)
            {
                //check if the amount i have is enough to complete the row
                if (amountOfColor + rowSqueresMatchingCondition >= currLevelData.Columns)
                {
                    //fix the list of potential indexes to only add the minimal amount needed

                    while (potentialIndexes.Count + rowSqueresMatchingCondition > currLevelData.Columns)
                    {
                        potentialIndexes.RemoveAt(0);
                    }

                    return potentialIndexes;
                }

            }
        }

        //if we got here - no valid rows were found

        potentialIndexes.Clear();

        amountOfColor = Math.Min(currLevelData.Rows - 1, _amountOfColor);

        //check cols
        for (int col = 0; col < currLevelData.Columns; col++)
        {
            potentialIndexes.Clear();
            int rowSqueresMatchingCondition = 0;
            int colColorIndex = colorIndex;
            bool colValid = true;

            for (int row = 0; row < currLevelData.Rows; row++)
            {
                int squereColorValue = gameBoard[row,col];

                //the row has at least 1 color that doesnt match
                if (squereColorValue != -1 && squereColorValue != colColorIndex)
                    colValid = false;

                if (squereColorValue == colColorIndex)
                    rowSqueresMatchingCondition++;

                //empty - add to list - maybe we can complete it
                if (squereColorValue == -1)
                    potentialIndexes.Add(new Vector2Int(row,col));
            }

            //row has 1 color or empty spaces
            if (colValid)
            {
                //check if the amount i have is enough to complete the row
                if (amountOfColor + rowSqueresMatchingCondition >= currLevelData.Rows)
                {
                    //fix the list of potential indexes to only add the minimal amount needed

                    while (potentialIndexes.Count + rowSqueresMatchingCondition > currLevelData.Rows)
                    {
                        potentialIndexes.RemoveAt(0);
                    }

                    return potentialIndexes;
                }

            }
        }

        potentialIndexes.Clear();
        return potentialIndexes;
    }

    private bool IsNoMoreMoves()
    {
        int numberOfShapesInWaitingArea = 0;
        int numberOfShapesThatCantBePlaced = 0;

        //need to loop over all the active shapes and see if none of them can fit anywhere
        foreach (ShapeView shape in _shapeStorage.shapeList)
        {            
            bool foundPositionForShape = false;

            numberOfShapesInWaitingArea++;

            for (int row = 0; row < currLevelData.Rows; row++)
            {
                for (int col = 0;col<currLevelData.Columns; col++)
                {
                    if (foundPositionForShape == false)
                    {
                        List<Vector2Int> shapesPositions = shape.GetShapeRelativeGridPositions();

                        bool shapeCanBePlacedInThisIndex = true;

                        //check if all the shape positions can be placed

                        //convert the vector positions to the array index

                        for (int j = 0; j < shapesPositions.Count; j++)
                        {
                            Vector2Int combinedPosition = new Vector2Int(row + shapesPositions[j].x, col + shapesPositions[j].y);

                            bool indexInGrid = true;

                            if (combinedPosition.x < 0 || combinedPosition.x > currLevelData.Rows-1)
                                indexInGrid = false;

                            if (combinedPosition.y < 0 || combinedPosition.y > currLevelData.Columns - 1)
                                indexInGrid = false;

                            if (indexInGrid == false || gameBoard[combinedPosition.x,combinedPosition.y] != -1)
                                shapeCanBePlacedInThisIndex = false;
                            
                        }

                        if (shapeCanBePlacedInThisIndex)
                        {
                            //found a valid place for the shape
                            foundPositionForShape = true;
                        }
                    }
                }
            }


            if (foundPositionForShape == false)
                numberOfShapesThatCantBePlaced++;
            
        }

        return numberOfShapesInWaitingArea == numberOfShapesThatCantBePlaced;
    }

using System;
using System.Collections.Generic;
using Newtonsoft.Json;


    /// <summary>
    /// Represents the data structure of a game level as defined in the JSON.
    /// </summary>
    public class LevelData
    {
    public LevelData(int rows, int columns)
    {
        Rows = rows;
        Columns = columns;
    }

    [JsonProperty("cols")]
        public int Columns { get; set; }

        [JsonProperty("rows")]
        public int Rows { get; set; }

        [JsonProperty("peopleQueues")]
        public List<List<int>> PeopleQueues { get; set; } = new List<List<int>>();

        [JsonProperty("shapesLines")]
        public List<ShapeInfo> ShapesLines { get; set; } = new List<ShapeInfo>();

        [JsonProperty("shapes_1_2")]
        public bool Shapes1And2 { get; set; }

        [JsonProperty("shapes_3")]
        public bool Shapes3 { get; set; }

        [JsonProperty("shapes_4")]
        public bool Shapes4 { get; set; }

        [JsonProperty("shapes_5")]
        public bool Shapes5 { get; set; }

        [JsonProperty("randomSeed")]
        public int RandomSeed { get; set; }

        [JsonIgnore]
        public int NumberOfColors { get; set; }




        /// <summary>
        /// Parses a JSON string and converts it into a LevelData object.
        /// </summary>

        /// <summary>
        /// Calculates the number of distinct colors from the people queues.
        /// </summary>
        /// <param name="peopleQueues">The list of people queues.</param>
        /// <returns>The count of distinct color indices.</returns>
        private static int CalculateNumberOfColors(List<List<int>> peopleQueues)
        {
            HashSet<int> distinctColors = new HashSet<int>();

            foreach (var queue in peopleQueues)
            {
                foreach (var color in queue)
                {
                    distinctColors.Add(color);
                }
            }

            return distinctColors.Count;
        }

public int[] GetQueueTop()
{
    int[] top = new int[numberOfColors];


    //top[0] - how many people of color index 0 are  

    for (int color = 0; color < numberOfColors; color++)
    {
        foreach (List<int> personInLine in currentLines)
        {
            bool colorChainBroken = false;

            for (int i = 0; i < personInLine.Count; i++)
            {
                if (personInLine[i] == color && colorChainBroken == false)
                    top[color]++;

                if(personInLine[i] != color)
                    colorChainBroken = true;
            }
        }
    }

    

    return top;
}
